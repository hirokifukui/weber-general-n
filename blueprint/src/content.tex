\chapter{Setting}
\begin{definition}[Weber layers, relative units, the KY unit, components]\label{def:setting}
$B_n=\mathbb{Q}(X_n)$, $X_n=2\cos(2\pi/2^{n+2})$; $G_n=\langle\sigma\rangle$, $\sigma(X_n)=2\cos(3\cdot 2\pi/2^{n+2})$;
$\tau=\sigma^{2^{n-1}}$; $E_n$ the units, $RE^+_n=\{\varepsilon: N_{n/n-1}\varepsilon=1\}$;
$\varepsilon_n=(X_n+1)/(X_n-1)$; $A_n=\langle\pm1,\varepsilon_n\rangle_{\mathbb{Z}[G_n]}$;
$m=2^{n-1}$; $R_n=\mathbb{Z}[x]/(x^m+1)$; $u_a=\prod\sigma^i(\varepsilon_n)^{a_i}$; $\lambda_j=\log|\sigma^j\varepsilon_n|$;
$H_n:\mathbb{Z}^m\to\mathbb{R}^{2m}$, $(H_na)_j=\sum_i a_i\lambda_{i+j}$; $D_n=\operatorname{covol}H_n(\mathbb{Z}^m)$;
$L_n=\sqrt{2^n}\log(2+\sqrt5)$; $d_n(\ell)=\operatorname{ord}_{2^n}(\ell)$; for $f\mid x^m+1$ irreducible over $\mathbb{F}_\ell$:
$M_f=((x^m+1)/f)\mathbb{F}_\ell[x]/(x^m+1)$, $L_f=\pi^{-1}(M_f)$, $\Lambda_f=\ell^{-1}H_n(L_f)$;
$C_n=(2/\pi)^{m/2}\Gamma(2+m/2)D_n/L_n^m$. The KY identification (KY sect.~4) is
\begin{equation}\label{eq:ident}
R_n/\ell R_n=\mathbb{F}_\ell[x]/(x^m+1)\;\xrightarrow{\ \sim\ }\;A_n^{1/\ell}/A_n,\qquad \bar a\mapsto r_aA_n,\quad r_a:=\prod_{i<m}\sigma^i(\varepsilon_n)^{a_i/\ell},
\end{equation}
$A_n^{1/\ell}=\{x\in\mathbb{R}:x^\ell\in A_n\}$, $r_a$ the real $\ell$-th root of $u_a$; well defined on classes mod $\ell$ since $a\mapsto a+\ell b$ multiplies $r_a$ by $u_b\in A_n$.
\end{definition}

\chapter{Literature inputs}
\begin{lemma}[KY index {[L]}]\label{lem:ky17}
$[RE^+_n:A_n]=h_n/h_{n-1}=:k_n$. (KY Eq.~(17), p.10, exact equality.)
\end{lemma}
\begin{lemma}[KY freeness and identification {[L]}]\label{lem:kyfree}
$A_n/\{\pm1\}$ is free abelian on $\sigma^i(\varepsilon_n)$, $0\le i<m$; $R_n/\ell R_n\simeq A_n^{1/\ell}/A_n$, $\bar a\mapsto r_aA_n$ with $r_a$ the real $\ell$-th root of $u_a$. (KY sect.~4, pp.13--14.)
\end{lemma}
\begin{lemma}[KY saturation {[L]}]\label{lem:ky41}
If $\ell\mid k_n$ then some irreducible $f_0\mid x^m+1$ over $\mathbb{F}_\ell$ has $\{g\cdot\varepsilon_n: g\bmod(x^m+1)\in M_{f_0}\}\subseteq RE^+_n$, i.e.\ the image of $M_{f_0}$ in $A_n^{1/\ell}/A_n$ lies in the subgroup $RE^+_n/A_n=\{[\varepsilon]:\varepsilon\in RE^+_n\cap A_n^{1/\ell}\}$ (a set of cosets). (KY Prop.~4.1, p.15, display (23).)
\end{lemma}
\begin{lemma}[KY trace floor {[L]}]\label{lem:ky23}
For $n\ge3$ and $\varepsilon\in RE^+_n\setminus\{\pm1\}$: $\operatorname{Tr}\varepsilon^2\ge33\cdot2^n$ (KY Thm.~2.3, p.4). For $n=2$: $\ge17\cdot2^n$ (MO3 Prop.~6.6).
\end{lemma}
\begin{lemma}[Height floor {[L]}]\label{lem:mo25-z2}
For $\varepsilon\in E_n\setminus E_{n-1}$ with $N_{n/n-1}\varepsilon=1$: $\operatorname{ht}(\varepsilon)\ge L_n$. (MO16 Lemma~2.5(1), p.817.)
\end{lemma}
\begin{lemma}[Blichfeldt {[L]}]\label{lem:blich}
A rank-$d$ lattice $\Lambda$ has a nonzero $v$ with
\[ |v|^2\le\frac{2}{\pi}\Gamma\big(1+\tfrac{d+2}{2}\big)^{2/d}\operatorname{covol}(\Lambda)^{2/d} \]
(Bl14 Thm.~II; MO16 Thm.~2.7). Equivalent power form: $|v|^d\le(2/\pi)^{d/2}\Gamma(2+d/2)\operatorname{covol}(\Lambda)$.
\end{lemma}
\begin{lemma}[Dirichlet {[L]}]\label{lem:dirichlet}
The logarithmic image of $E_n$ in $\mathbb{R}^{2^n}$ is discrete; $L(1,\chi)\ne0$ for nontrivial Dirichlet $\chi$.
\end{lemma}
\begin{lemma}[Schinzel's inequality {[L]}]\label{lem:schinzel}
Let $\varepsilon$ be a totally real unit different from $\pm1$ and $\mathfrak M$ an ideal of $\mathbb Q(\varepsilon)$ containing $\varepsilon^2-1$, $C$ the absolute norm of $\mathfrak M$, $d=[\mathbb Q(\varepsilon):\mathbb Q]$. Then $M(\varepsilon)\ge\big((C^{1/d}+\sqrt{C^{2/d}+4})/2\big)^{d/2}$. (MO3 Thm.~5.1, p.~1074, quoting Schinzel; verbatim in paper/sources/MO3\_2020\_clean.txt, line 631.)
\end{lemma}
\begin{lemma}[Height versus Mahler measure {[L]}]\label{lem:mo24}
For a unit $\varepsilon$ of a totally real field of degree $N$: $\log M(\varepsilon)\le\tfrac12\sqrt N\operatorname{ht}(\varepsilon)$ (MO16 eq.~(2.4), proof of Lemma~2.3; Cauchy--Schwarz, reproved inside Lemma~\ref{lem:depthfloor}).
\end{lemma}
\begin{lemma}[Ramar\'e's bound for $|L(1,\chi)|$, even conductor {[L]}]\label{lem:ramare}
For every even primitive Dirichlet character $\chi$ of conductor $q$ with $2\mid q$ and $q\ge4$: $|L(1,\chi)|\le U(q):=\frac14(\log q+2\log2)+\log(4q)/\sqrt q$. (Ram04 Corollary~1 with $h=1$, $k=2$: $\omega(h)=0$, $\phi(hk)/(2hk)=1/4$, $\sum_{p\mid 2}\log p/(p-1)=\log2$, $\kappa_\chi=0$ for even $\chi$, error term $\phi(h)2^{\omega(k)-1}h^{-1}q^{-1/2}\log(q\cdot4)$; condition $q\ge k^24^{\omega(h)}=4$. Read from the author's accepted manuscript; the journal typesetting is to be compared before release.)
\end{lemma}
\begin{lemma}[Robbins' Stirling bound {[L]}]\label{lem:robbins}
For every integer $k\ge1$: $k!=\sqrt{2\pi k}\,(k/e)^k e^{r_k}$ with $1/(12k+1)<r_k<1/(12k)$; in particular $k!\le\sqrt{2\pi k}\,(k/e)^ke^{1/(12k)}$. (Rob55.)
\end{lemma}
\begin{lemma}[Horie's saturation lemma at odd $p$ {[L]}; Ho05a Prop.~1 read in r17]\label{lem:horie13}
$p$ odd, $\ell\ne p$ prime, $F\subseteq\Q(\zeta_{p^n})$ an extension of the decomposition field of $\ell$; $\eta_n$ the Horie unit; $\alpha\mapsto\alpha_\sigma$ the coordinate lift $\Z[\zeta_{p^n}]\to\Z[\operatorname{Gal}]$. Then $\ell\mid h_{p,n}/h_{p,n-1}$ iff there is a prime $\mathcal L$ of $F$ above $\ell$ with $\eta_n^{\alpha_\sigma}$ an $\ell$-th power in $E_{p,n}$ for every $\alpha\in\ell\mathcal L^{-1}$. (MO13 Lemma~1.3, attributed to Ho05b, which relays Ho02 Lemmas 2, 3, 8; the same statement is quoted in Horie--Horie, Tohoku 61 (2009). Only the direction ``$\ell\mid h_{p,n}/h_{p,n-1}\Rightarrow$'' is used. r17: the general form of the lemma, with proof, is Ho05a Proposition~1 (open access; paper/Horie2005\_JMSJ\_triviality.pdf, read): for a Dirichlet character $\chi$ of order $g_\chi$ prime to $\ell$, $\sigma$ a generator of $\operatorname{Gal}(K_\chi/\Q)$ and $k\subseteq\Q(\zeta_{g_\chi})$ containing the decomposition field of $\ell$, $\ell\mid\bar h_\chi$ iff there is a prime $\mathfrak l$ of $k$ above $\ell$ such that $|\eta_\chi^{\alpha_\sigma}|$ is an $\ell$-th power in $E_\chi$ for every $\alpha\in\ell\mathfrak l^{-1}$; MO13 Lemma~1.3 is its specialisation to the layer ($K_\chi=B_{p,n}$, $\bar h_\chi=h_{p,n}/h_{p,n-1}$, $\eta_\chi$ the Horie unit up to sign and a root of unity). The original Ho02 (JLMS, paywalled) has still not been compared verbatim; Ho05a is by the same author and contains the proof.)
\end{lemma}
\par\noindent{\small\emph{Historical / superseded input (not a node of the graph; used by no current theorem).} The r16 floor was the Schinzel-type Mahler-measure bound $M(\varepsilon)\ge((1+\sqrt5)/2)^{d/2}$ for a totally real unit $\varepsilon\ne\pm1$ of degree $d$ (MO13 Thm.~2.2 with the ideal $\mathfrak M=(1)$; Sch73); superseded in r17 by Lemma~\ref{lem:mo25-z3} and kept here only for the record (1.0.1, GPT r21 item 8).}\par
\begin{lemma}[Washington {[L]}]\label{lem:wash}
For $\chi$ even primitive of conductor $f>1$:
\[ L(1,\chi)=-\frac{\tau(\chi)}{f}\sum_{a=1}^f\bar\chi(a)\log|1-\zeta_f^a| ,\qquad |\tau(\chi)|=\sqrt f . \] (Wash, Chapter~4: the $L(1,\chi)$ formula for even primitive $\chi$ and the Gauss-sum modulus; cited at chapter level, hw 29.)
\end{lemma}
